  email : VARCHAR

}

  --
  name : VARCHAR
}

entity user_roles {
  + user_id : INT <<PK, FK>>

}

entity user_progress {
  + user_id : INT <<PK, FK>>
  --
  level : INT
  experience : INT
}

entity artists {
  + id : INT <<PK>>
  --
  name : VARCHAR
}

entity albums {
  + id : INT <<PK>>
  --
  title : VARCHAR
  artist_id : INT <<FK>>
}

entity songs {
  + id : INT <<PK>>
  --
  title : VARCHAR
  artist_id : INT <<FK>>
  album_id : INT <<FK>>
}

entity playlists {
  + id : INT <<PK>>
  --
  name : VARCHAR
  user_id : INT <<FK>>
}

entity playlist_songs {
  + playlist_id : INT <<PK, FK>>
  + song_id : INT <<PK, FK>>
}

entity missions {
  + id : INT <<PK>>
  --
  title : VARCHAR
  description : TEXT
  reward_xp : INT
}

entity user_missions {
  + user_id : INT <<PK, FK>>

  --
  status : VARCHAR
  completed_at : TIMESTAMP
}

entity achievements {
  + id : INT <<PK>>
  --
  name : VARCHAR
  description : TEXT
}

  + user_id : INT <<PK, FK>>
  + achievement_id : INT <<PK, FK>>
  --
  unlocked_at : TIMESTAMP
}

entity collectibles {
  + id : INT <<PK>>
  --
  name : VARCHAR
  type : VARCHAR
  rarity : VARCHAR
}

entity user_collectibles {
  + user_id : INT <<PK, FK>>
  + collectible_id : INT <<PK, FK>>
  --
  acquired_at : TIMESTAMP
}

  + id : INT <<PK>>
  --
  title : VARCHAR
  narrative_text : TEXT
}

entity user_story_progress {
  + user_id : INT <<PK, FK>>
  + story_node_id : INT <<PK, FK>>
  --
  choice_taken : VARCHAR
